02. CMakeLists.txt 작성하기#

  1. root 폴더에 CMakeLists.txt 파일 생성 
    cmake_minimum_required(VERSION 3.14)
project([[게임이름]] C CXX)

# raylib 불러오는 코드
include(FetchContent)
FetchContent_Declare(
    raylib
    GIT_REPOSITORY https://github.com/raysan5/raylib.git
    GIT_TAG        6.0 # 원하는 버전 tag 지정 (예: 5.0, master 등)
)
FetchContent_MakeAvailable(raylib)

# 빌드 옵션
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE) # 예제 빌드 비활성화

# 실행파일 생성
add_executable(${PROJECT_NAME} [[메인파일 ex) src/main.cpp]])

# raylib 타켓 링크
target_link_libraries(${PROJECT_NAME} PRIVATE raylib)

03. hello world 추가하기#

  1. root/src/main.cpp 작성하기 
    #include "raylib.h"

int main() {
    // 1. 창 크기와 제목 설정 후 초기화
    const int screenWidth = 800;
    const int screenHeight = 450;
    InitWindow(screenWidth, screenHeight, "raylib - Hello World");

    // 2. 초당 프레임 수(FPS)를 60으로 설정
    SetTargetFPS(60);

    // 3. 메인 게임 루프 (ESC 키나 닫기 버튼을 누르면 종료)
    while (!WindowShouldClose()) {
        // --- 렌더링 시작 ---
        BeginDrawing();

        // 배경색을 흰색으로 칠하기
        ClearBackground(RAYWHITE);

        // 텍스트 그리기: (내용, X좌표, Y좌표, 폰트크기, 색상)
        // MeasureText()를 활용해 텍스트를 정중앙에 배치합니다.
        int textWidth = MeasureText("Hello World!", 40);
        DrawText("Hello World!", (screenWidth - textWidth) / 2, (screenHeight - 40) / 2, 40, DARKGRAY);

        // --- 렌더링 종료 ---
        EndDrawing();
    }

    // 4. 창을 닫고 리소스 해제
    CloseWindow();

    return 0;
}
